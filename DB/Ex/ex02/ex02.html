Exercise 2.2 [Defining Database Schemas] 

CREATE TABLE person (
    driver_id VARCHAR(15),
    name VARCHAR(63) NOT NULL,
    address VARCHAR(127) UNIQUE NOT NULL,
    PRIMARY KEY (driver_id)
);

CREATE TABLE car (
    license VARCHAR(),
    model VARCHAR(127) NOT NULL,
    year INTEGER NOT NULL DEFAULT CURRENT_DATE,
    PRIMARY_KEY (licence)
    CHECK (year >= 1850 AND year <= YEAR(CURRENT_DATE))
);

CREATE TABLE accident (
    report_number INT(15),
    date DATE NOT NULL,
    location VARCHAR(127) NOT NULL,
    PRIMARY_KEY (report_number)
);

CREATE TABLE owns (
    driver_id VARCHAR(15) NOT NULL,
    license VARCHAR(15) NOT NULL,
    PRIMARY KEY (driver_id, license),
    FOREIGN KEY (driver_id) REFERENCES person(driver_id)
        ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (license) REFERENCES car(license)
        ON DELETE CASCADE ON UPDATE CASCADE
);

CREATE TABLE participated (
    report_number INT(15) AUTO_INCREMENT,
    license VARCHAR(15) NOT NULL,
    driver_id VARCHAR(15) NOT NULL,
    damage_amount DECIMAL(10, 2) NOT NULL,
    PRIMARY KEY (report_number, license, driver_id),
    FOREIGN KEY (report_number) REFERENCES accident(report_number)
        ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (license) REFERENCES car(license)
        ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (driver_id) REFERENCES person(driver_id)
        ON DELETE CASCADE ON UPDATE CASCADE,
    CHECK (damage_amount >= 0)
);
