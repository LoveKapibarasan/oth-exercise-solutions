\section*{5. Independent events and independent repeated trials}

\subsection* {E16}
The sample space of A and B are independent.
\begin{enumerate}
  \item $P(A \cap B) = P_A(Red) \cdot P_B(red) = P(C)  =  \underline{\underline{ \frac{3}{8} }} $
  \item $P(\text{Red and Blue } = P_A(\text{Red}) \cdot P_B(\text{Blue}) +   P_B(\text{Red}) \cdot P_A(\text{Blue}) =  \underline{\underline{ \frac{19}{40} }} $
\end{enumerate}

\subsection* {E17}

\subsubsection*{case 1:}
\begin{itemize}
\item The sample space = \{BBB, BBG, BGB, BGG, GBB, GBG, GGB, GGG\}
\item A (both sexes) = \{BBG, BGB, BGG, GBB, GBG, GGB \}
\item B (at most one boy) = \{BGG, GBG, GGB, GGG\}
\item $A \cap B$ \{ BGG, GBG, GGB \}
\end{itemize}

\subsubsection*{case 2:}
\begin{itemize}
\item The sample space = \{BB, BG, GB, GG\}
\item A (both sexes) = \{BG, GB \}
\item B (at most one boy) = \{BG, GB, GG \}
\item $A \cap B$ \{ BG, GB \}
\end{itemize}

\begin{enumerate}
  \item $ \underline{\underline{ \text{Independent because }  }} P(A) \cdot P(B) = P(A \cap B) = \frac{3}{8} $ 
  \item $ \underline{\underline{ \text{Dependent because }  }} P(A) \cdot P(B) \neq P(A \cap B) $
\end{enumerate}
