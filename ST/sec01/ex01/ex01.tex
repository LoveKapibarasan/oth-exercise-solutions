\section {1. Counting techniques, sample spaces and events}

\subsection* {T1}
\begin{enumerate}
  \item \underline{\underline{14\text{ combinations.}}} 
  \item $ 8 \times 6 = \underline{\underline{48\text{ combinations.}}} $
  \item $ 14 \times (14 - 1) = \underline{\underline{163\text{ combinations.}}}$
\end{enumerate}

\subsection* {T2} 
\begin{enumerate}
  \item \underline{\underline{20\text{ combinations.}}} 
  \item $20 \times 20 = \underline{\underline{400\text{ combinations.}}}$
  \item $400 -  5(4 \times 3) - 4(5 \times 4) - 5 \times 4 = \underline{\underline{240\text{ combinations.}}}$
\end{enumerate}

\subsection* {T3} 
\textbf{Sequence:}
\begin{enumerate}
  \item $5! = \underline{\underline{120\text{ combinations.}}} $
  \item $2! ( 2! \times 3!) = \underline{\underline{24\text{ combinations.}}}$
\end{enumerate}

\subsection* {T4} 
\textbf{Sequence:}
\begin{enumerate}
  \item $8 ^ 3 = \underline{\underline{512\text{ combinations.}}} $
  \item $ 8 \times 7 \times 6 = \underline{\underline{336\text{ combinations.}}}$
\end{enumerate}

\subsection* {T5} 
\textbf{Combination:}
\begin{enumerate}
  \item $ C(12, 4) = \underline{\underline{495\text{ combinations.}}} $
\end{enumerate}

\subsection* {T6} 
\textbf{Combination: dodecagon}
\begin{enumerate}
  \item $C(12, 2) = \underline{\underline{66\text{ combinations.}}} $
  \item \underline{\underline{11\text{ combinations.}}} 
  \item $ C(11, 2) = \underline{\underline{55\text{ combinations.}}} $
\end{enumerate}

\subsection* {T6} 
\textbf{Random walk:}
\begin{enumerate}

  \item $\underline{\underline{2\text{ combinations.}}} $
  \item $ 1 + 1 + 2 = \underline{\underline{4\text{ combinations.}}} $
\end{enumerate}
