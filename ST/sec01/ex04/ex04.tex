\section*{4. Law of total probability and Bayes’ theorem}
\subsection* {E12}
\begin{enumerate}
  \item $P(V) = P(C) \cdot P(V|C) + P(L) \cdot P(V|L) + P(I) \cdot P(V|I) =  \underline{\underline{ 0.47 }} $
  \item 
  	\begin{itemize}
  	\item $P(C|V) = \frac{ P(C) \cdot P(V|C) }{ P(V)} =  \underline{\underline{ \frac{18}{47} }} $
  	\item $P(V|V) =   \underline{\underline{ \frac{14}{47} }} $
  	\item $P(I|V) =   \underline{\underline{ \frac{15}{47} }} $
	\end{itemize}
\end{enumerate}
