\section* {3. Joint distribution and independent random variable}

\subsection*{Covariance}

In probability space $(S, E_x, f_x)$ and $(S, E_y, f_y)$, 
\begin{align*}
X &: \{s_x\} \in S \rightarrow X(s_x) \\
Y &: \{s_y\} \in S \rightarrow Y(s_y) \\
XY &:  \{ (s_x, s_y)\} \in S \rightarrow X(s_x) \cdot Y(s_y) \\
Z = (X-\mu_X)(Y - \mu_Y) &: \{ (s_x, s_y) \} \in S \rightarrow (X(s_x) - \mu_X)(Y(s_y) - \mu_Y) \\
\text{cov}(X,Y) &= E[Z] \\
\rho(X,Y) &= \frac{\text{cov}(X,Y)}{\sigma_X \cdot \sigma_Y}
\end{align*}
