\begin{align*}
E(X) &= \sum_{s_i \in S}^{ } x_i \cdot P(x_i) \\
     &= \sum_{i = 0 }^{ \infty } i \cdot q^{i-1} p \\

     &= \frac{1}{p}
\end{align*}
