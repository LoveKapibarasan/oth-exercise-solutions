\subsection* {T7}
\begin{align*}
S_\text{original} &=  \{ 1,2,3,4,5,6 \}  \\
X &: \{ x \} \in S \rightarrow 2x ( \in \{ 2,4,6,8,10,12 \} ) (\text{ each probability is }\frac{1}{6}  ) \\
X^2 &: \{ y \} \in X(S) \rightarrow y^2 ( \in \{ 4,16,36,64,100,144 \}) (\text{ each probability is }\frac{1}{6} \text{ and } y \in \mathbb{R} \\
& \text{ and I can regard this as randam function(variable)  } \\
\mu_X = E[X] &= \frac{2+4+6+8+10+12}{6} = 7 \\
E[ X^2 ] &= \frac{2^2+4^2+6^2+8^2+10^2+12^2}{6} = \frac{182}{3} \\
\sigma_X^2 &= E[X^2] - (E[X])^2 = \frac{182}{3} - 49 = \frac{182 - 147}{3} = \boxed{\frac{35}{3}} \\
\sigma_X &= \sqrt{\frac{35}{3}} = \boxed{\frac{\sqrt{105}}{3}}
\end{align*}
