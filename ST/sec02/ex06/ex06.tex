\[
f_X(x) =
\begin{cases}
\dfrac{1}{b-a}, & a \le x \le b, \\[2mm]
0, & \text{otherwise}.
\end{cases}
\]

\textbf{1. }
\[
E[X] = \int_{-\infty}^{\infty} x f_X(x) \, dx
= \int_a^b x \cdot \frac{1}{b-a} \, dx
= \frac{1}{b-a} \int_a^b x \, dx = \boxed{E[X] = \frac{a+b}{2}}
\]

\textbf{2. }
\[
E[X^2] = \int_a^b x^2 \cdot \frac{1}{b-a} \, dx
= \frac{1}{b-a} \int_a^b x^2 \, dx
= \frac{1}{b-a} \cdot \frac{b^3 - a^3}{3}
= \frac{b^2 + ab + a^2}{3}.
\]

\[
\boxed{\mathrm{Var}(X) = \frac{(b-a)^2}{12}}
\]

\textbf{3. }
\[
F_X(x) = P(X \le x) = \int_{-\infty}^x f_X(t) \, dt
=
\begin{cases}
0, & x < a, \\
\dfrac{x-a}{b-a}, & a \le x \le b, \\
1, & x > b.
\end{cases}
\]
