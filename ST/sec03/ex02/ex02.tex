\section*{2. The Central Limit Theorem and Applications}

\subsection*{T4}

\textbf{1.}

Let $X \sim \text{Bin}(12, 0.5)$, then
\begin{align*}
P(4 \leq X \leq 7) &= \sum_{k=4}^{7} \binom{12}{k} \left(\frac{1}{2}\right)^{12} = \frac{3003}{4096} \approx 0.733
\end{align*}

\textbf{2.}

Let $X \approx N(\mu = 6, \sigma^2 = 3)$

\textbf{Continuity Correction:} $X = k \Leftrightarrow k - 0.5 \leq Y \leq k + 0.5$

Then, $P(4 \leq X \leq 7) \approx P(a_1 = 3.5 \leq Y \leq a_2 = 7.5)$

Normalization: $z_1 = -1.443$ for $a_1$ and $z_2 = 0.866$ for $a_2$

Therefore,
\begin{align*}
P(4 \leq X \leq 7) &\approx \Phi(|z_1|) + \Phi(|z_2|) \approx 0.732
\end{align*}

\subsection*{Ex3}

\begin{align*}
\mu = E[X] &= \int_{-\infty}^{\infty} x \cdot 3(1 - x)^2 \, dx = \frac{1}{4} \\
E[X^2] &= 0.1 \\
\text{Var}(X) &= \sigma^2 = 0.0375
\end{align*}

Let
\begin{align*}
S &= X_1 + \cdots + X_{625} \\
E[S] &= 625\mu = 156.25 \quad \text{(Linearity)} \\
\text{Var}(S) &= 625\sigma^2 = 23.4375
\end{align*}

Thus by the Central Limit Theorem,
\begin{align*}
S &\approx N(E[S], \text{Var}(S))
\end{align*}

$P(S \leq 170 = a_2)$

Normalization: $z_2 = 2.84$ for $a_2$

Therefore,
\begin{align*}
P(S \leq 170) &\approx \Phi(|z_2|) \approx 0.9977
\end{align*}
