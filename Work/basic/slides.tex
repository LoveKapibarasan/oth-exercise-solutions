\section{Payment Modalities}

\begin{frame}{Core Payment Modalities}
    \begin{small}
    \begin{enumerate}[label=\protect\circlebullet{\arabic*}, leftmargin=0.8cm, labelsep=0.3cm, itemsep=0.1cm]
        \item \textbf{QR Code (Scan \& Charge)} -- \textit{Target: Guest Users} \\
              Instant web-based payment flow. No app installation or registration required.
        \item \textbf{NFC Sticker (Scan \& Charge)} -- \textit{Target: Guest Smartphone Users} \\
              High-speed "Tap to Start" interaction via passive NDEF-formatted tags.
        \item \textbf{RFID (Subscription)} -- \textit{Target: Fleet / Monthly Users} \\
              Dedicated physical cards optimized for subscription-based flat-rate plans.
        \item \textbf{Touch \& Charge (Stripe Terminal)} -- \textit{Target: Credit Card Holders} \\
              Physical hardware integration supporting EMV contactless, Apple/Google Pay.
        \item \textbf{Plug \& Charge (Hubject)} -- \textit{Target: ISO 15118 Compatible EVs} \\
              Automated authentication and billing directly handled through the charging cable.
    \end{enumerate}
    \end{small}
\end{frame}
