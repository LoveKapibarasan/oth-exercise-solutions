def sos2(a, b):
m = len(a)
# Initialize an (m+1) x (b+1) matrix with 0
M = [[0 for j in range(b + 1)] for i in range(m + 1)]
M[0][0] = 1
for i in range(1, m + 1):
for j in range(b + 1):
